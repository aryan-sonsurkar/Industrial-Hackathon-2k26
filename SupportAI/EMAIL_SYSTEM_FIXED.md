try:
    # Email sending logic
    server = smtplib.SMTP(SMTP_SERVER, SMTP_PORT)
    server.starttls()
    server.login(SMTP_USERNAME, SMTP_PASSWORD)
    server.send_message(msg)
    server.quit()
    return {"status": "success", "message": "Email sent successfully"}

except smtplib.SMTPAuthenticationError as e:
    return {"status": "failed", "message": "Email authentication failed"}

except smtplib.SMTPRecipientsRefused as e:
    return {"status": "failed", "message": "Email server refused recipient"}

except smtplib.SMTPServerDisconnected as e:
    return {"status": "failed", "message": "Email server disconnected"}

except Exception as e:
    return {"status": "failed", "message": "Email could not be sent"}
